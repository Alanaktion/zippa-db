SELECT u.id, u.email, u.role, COUNT(o.id) AS total_orders, MAX(o.created_at) AS last_active
FROM users u
LEFT JOIN orders o ON o.user_id = u.id
WHERE u.status = 'active'
GROUP BY u.id, u.email
ORDER BY total_orders DESC
LIMIT 100;
